123constexpr uint8_t Crc8(uint8_t init, uint8_t high, uint8_t low) noexcept {
124 uint32_t crc = init; // all arithmetic in uint32_t (KB-CPP20-021)
125 for (uint32_t byte = 0U; byte < 2U; ++byte) {
126 crc ^= (byte == 0U) ? static_cast<uint32_t>(high) : static_cast<uint32_t>(low);
127 for (uint32_t bit = 0U; bit < 8U; ++bit) {
128 crc = ((crc & 0x80U) != 0U) ? ((crc << 1U) ^ 0x31U) : (crc << 1U);
129 crc &= 0xFFU;
130 }
131 }
132 return static_cast<uint8_t>(crc);
133}

140[[nodiscard]] constexpr bool CrcVectorsHold() noexcept {
141 using detail::operator""_u8;
142 // CRC(0xBEEF) = 0x92 with initial value 0xFF (every SF06 document, Table 14).
143 bool ok = Crc8(0xFF_u8, 0xBE_u8, 0xEF_u8) == 0x92_u8;
144 // The same word with the SF05 initial value 0x00 (AN-SFM-04): 0x13.
145 ok = ok && (Crc8(0x00_u8, 0xBE_u8, 0xEF_u8) == 0x13_u8);
146 // The SFx6000 worked sequence `36 61 36 08 D0`: CRC(0x3608) = 0xD0.
147 ok = ok && (Crc8(0xFF_u8, 0x36_u8, 0x08_u8) == 0xD0_u8);
148 // The SFM3304 scale-read argument 0x3603 carries CRC 0x3A (SFM3304
149 // section 4.3.5).
150 ok = ok && (Crc8(0xFF_u8, 0x36_u8, 0x03_u8) == 0x3A_u8);
151 // A zero word is not a valid frame with the SF06 initial value: a stuck-low
152 // bus fails its CRC.
153 ok = ok && (Crc8(0xFF_u8, 0x00_u8, 0x00_u8) == 0x81_u8);
154 // A zero word with the SF05 initial value is valid (CRC 0x00), so SF05 needs
155 // a reply check too.
156 return ok && (Crc8(0x00_u8, 0x00_u8, 0x00_u8) == 0x00_u8);
157}
158static_assert(CrcVectorsHold());
