39// Optional hooks: a `Declares*` concept only tests the name, the `Has*` concept
40// the exact signature, `noexcept` and the result type; the driver rejects a
41// hook that is declared but does not match (KB-CPP20-007).

101template <typename Port>
102void DelayUs(Port& port, uint32_t us) noexcept {
103 if constexpr (HasDelayUs<Port>) {
104 port.DelayUs(us);
105 } else {
