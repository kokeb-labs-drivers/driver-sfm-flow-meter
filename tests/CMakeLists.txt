# =============================================================================
# SFM_FLOW_METER driver: host tests
# =============================================================================
# The driver against a register model of the chip, with the firmware's flags
# (-fno-exceptions -fno-rtti) and warnings as errors. No hardware, no ESP-IDF:
#
#   cmake -S . -B build -DSFM_FLOW_METER_BUILD_TESTS=ON
#   cmake --build build
#   ctest --test-dir build --output-on-failure
# =============================================================================

# ISO C++20, never the GNU dialect (handbook KB-CPP20-001): set here too, so that this folder is strict
# whoever includes it.
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# -pedantic-errors and the C++20 deprecations as errors (KB-CPP20-001, -003). -Wswitch-enum: a switch over
# an enumeration ends with `default: break;` (MISRA C++:2023 Rule 9.4.2), and a new enumerator still warns.
set(SFM_FLOW_METER_TEST_FLAGS -fno-exceptions -fno-rtti -Wall -Wextra -Wpedantic -pedantic-errors -Wshadow
    -Wconversion -Wsign-conversion -Wdeprecated -Wdeprecated-enum-enum-conversion
    -Wdeprecated-enum-float-conversion -Wimplicit-fallthrough -Wswitch-enum -Werror)
# The standard library's own bounds checks, so an out-of-range span or array access traps in the tests
# and fuzzers (KB-CPP20-043).
set(SFM_FLOW_METER_TEST_DEFINITIONS _GLIBCXX_ASSERTIONS=1 _LIBCPP_HARDENING_MODE=_LIBCPP_HARDENING_MODE_FAST)

function(sfm_flow_meter_test_target name)
  target_link_libraries(${name} PRIVATE sfm_flow_meter::sfm_flow_meter)
  target_include_directories(${name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  target_compile_features(${name} PRIVATE cxx_std_20)
  target_compile_options(${name} PRIVATE ${SFM_FLOW_METER_TEST_FLAGS})
  target_compile_definitions(${name} PRIVATE ${SFM_FLOW_METER_TEST_DEFINITIONS})
endfunction()

# --- Strict-dialect header checks ---------------------------------------------
# One translation unit per public header (and through the driver header its src/ .ipp): each header
# compiles on its own, first in its unit, as ISO C++20 with -pedantic-errors, and the unit asserts
# that the dialect really is C++20 (handbook KB-CPP20-001, checks strict-dialect and
# header-self-contained). They build with the host tests on GCC and Clang and, as part of
# sfm_flow_meter_optimised_build, in the Cortex-M cross-build.
file(GLOB_RECURSE SFM_FLOW_METER_PUBLIC_HEADERS CONFIGURE_DEPENDS ${PROJECT_SOURCE_DIR}/inc/*.hpp
  ${PROJECT_SOURCE_DIR}/inc/*.h)
set(SFM_FLOW_METER_HEADER_CHECKS)
foreach(header ${SFM_FLOW_METER_PUBLIC_HEADERS} ${SFM_FLOW_METER_VERSION_HEADER})
  if(header STREQUAL SFM_FLOW_METER_VERSION_HEADER)
    file(RELATIVE_PATH rel ${SFM_FLOW_METER_VERSION_HEADER_DIR} ${header})
  else()
    file(RELATIVE_PATH rel ${PROJECT_SOURCE_DIR}/inc ${header})
  endif()
  string(MAKE_C_IDENTIFIER "${rel}" id)
  set(check ${CMAKE_CURRENT_BINARY_DIR}/header_checks/${id}.cpp)
  string(CONCAT content "// Generated by tests/CMakeLists.txt: ${rel} compiles on its own as ISO C++20.\n"
    "#include \"${rel}\"\n"
    "static_assert(__cplusplus == 202002L, \"the strict-dialect check must build as -std=c++20\");\n")
  if(EXISTS ${check})
    file(READ ${check} old)
  else()
    set(old "")
  endif()
  if(NOT old STREQUAL content)
    file(WRITE ${check} "${content}")
  endif()
  list(APPEND SFM_FLOW_METER_HEADER_CHECKS ${check})
endforeach()
add_library(sfm_flow_meter_header_checks OBJECT ${SFM_FLOW_METER_HEADER_CHECKS})
sfm_flow_meter_test_target(sfm_flow_meter_header_checks)

# --- Version: the getters, and every copy of the version string ---------------
function(sfm_flow_meter_version_test name include_dir major minor patch)
  add_executable(${name} test_version.cpp)
  if(include_dir)
    target_include_directories(${name} BEFORE PRIVATE ${include_dir})
  endif()
  sfm_flow_meter_test_target(${name})
  target_compile_definitions(${name} PRIVATE EXPECTED_MAJOR=${major} EXPECTED_MINOR=${minor}
    EXPECTED_PATCH=${patch} EXPECTED_STRING="${major}.${minor}.${patch}")
  add_test(NAME ${name} COMMAND ${name})
endfunction()

sfm_flow_meter_version_test(sfm_flow_meter_version "" ${PROJECT_VERSION_MAJOR} ${PROJECT_VERSION_MINOR}
  ${PROJECT_VERSION_PATCH})

# The same template with another version, found ahead of the real header.
function(sfm_flow_meter_version_probe_header)
  set(SFM_FLOW_METER_VERSION_MAJOR 9)
  set(SFM_FLOW_METER_VERSION_MINOR 8)
  set(SFM_FLOW_METER_VERSION_PATCH 7)
  set(SFM_FLOW_METER_VERSION 9.8.7)
  configure_file(${SFM_FLOW_METER_VERSION_TEMPLATE} ${CMAKE_CURRENT_BINARY_DIR}/version_probe/sfm_flow_meter_version.h @ONLY)
endfunction()
sfm_flow_meter_version_probe_header()
sfm_flow_meter_version_test(sfm_flow_meter_version_probe ${CMAKE_CURRENT_BINARY_DIR}/version_probe 9 8 7)

# --- Compile-time contract: these must fail to build, for the reason in their EXPECT: line ---
file(GLOB SFM_FLOW_METER_COMPILE_FAIL CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/compile_fail/*.cpp)
foreach(src ${SFM_FLOW_METER_COMPILE_FAIL})
  get_filename_component(stem ${src} NAME_WE)
  file(STRINGS ${src} expect REGEX "EXPECT: ")
  string(REGEX REPLACE ".*EXPECT: " "" expect "${expect}")
  add_executable(sfm_flow_meter_compile_fail_${stem} EXCLUDE_FROM_ALL ${src})
  sfm_flow_meter_test_target(sfm_flow_meter_compile_fail_${stem})
  add_test(NAME sfm_flow_meter_compile_fail_${stem}
    COMMAND ${CMAKE_COMMAND} --build ${CMAKE_BINARY_DIR} --target sfm_flow_meter_compile_fail_${stem})
  # Pass only when the compiler rejects the code with the expected diagnostic, not another error.
  set_tests_properties(sfm_flow_meter_compile_fail_${stem} PROPERTIES PASS_REGULAR_EXPRESSION "${expect}")
endforeach()
